cmake_minimum_required(VERSION 3.20)
project(ScrollSpeed CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG v1.3.4)
FetchContent_MakeAvailable(minhook)
add_library(ScrollSpeed SHARED source/ScrollSpeed.cpp)
target_include_directories(ScrollSpeed PRIVATE ${minhook_SOURCE_DIR}/include)
target_link_libraries(ScrollSpeed PRIVATE minhook)
set_target_properties(ScrollSpeed PROPERTIES PREFIX "" SUFFIX ".asi")
